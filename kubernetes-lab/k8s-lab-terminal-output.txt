==================== SETUP: build the cluster ====================

ayush@ubuntu:~/k8s-lab$ {
  echo 'kind: Cluster'
  echo 'apiVersion: kind.x-k8s.io/v1alpha4'
  echo 'name: k8s-lab'
  echo 'nodes:'
  echo '  - role: control-plane'
  echo '    extraPortMappings:'
  for p in $(seq 30000 30050); do
    echo "      - containerPort: $p"
    echo "        hostPort: $p"
    echo "        protocol: TCP"
  done
  echo '  - role: worker'
  echo '  - role: worker'
} > kind-config.yaml
